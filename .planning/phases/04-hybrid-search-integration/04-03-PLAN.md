    ```rust
    #[tracing::instrument(skip(self, request), fields(collection_name, user_count))]
    async fn hybrid_search(
        &self,
        request: tonic::Request<HybridSearchRequest>,
    ) -> Result<tonic::Response<HybridSearchResponse>, tonic::Status> {
        let req = request.into_inner();
        let collection_name = req.collection_name;
        let user_ids = ids_to_u128s(&req.user_ids)?;
        
        // Validate inputs
        let has_text = req.text_query.is_some() && !req.text_query.as_ref().unwrap().is_empty();
        let has_vector = !req.vector.is_empty();
        
        if !has_text && !has_vector {
            return Err(tonic::Status::invalid_argument(
                "Either text_query or vector must be provided"
            ));
        }
        
        // Clamp alpha to [0, 1]
        let alpha = req.alpha.clamp(0.0, 1.0);
        
        // Handle query mode
        let mode = req.mode;
        let effective_alpha = match mode {
            HybridQueryMode::TextOnly => 1.0,
            HybridQueryMode::VectorOnly => 0.0,
            HybridQueryMode::Hybrid => alpha,
        };
        
        // Get collection and execute hybrid search
        if let Some(collection) = self.collection_manager.read().await.get_collection(&collection_name).await {
            let snapshot = collection.get_snapshot().await
                .map_err(|e| tonic::Status::internal(format!("Failed to get snapshot: {}", e)))?;
            
            // Execute hybrid search for all users
            let mut all_results = vec![];
            for user_id in user_ids {
                let results = collection.search_hybrid(
                    user_id,
                    req.text_query.clone().unwrap_or_default(),
                    req.vector.clone(),
                    req.params.as_ref().map(|p| p.top_k as usize).unwrap_or(10),
                    effective_alpha,
                ).await;
                
                if let Ok(user_results) = results {
                    all_results.extend(user_results);
                }
            }
            
            // Deduplicate and sort
            // ... (similar to search() handler)
            
            Ok(tonic::Response::new(HybridSearchResponse {
                doc_ids: all_results.iter().map(|(id, _)| u128_to_id(*id)).collect(),
                scores: all_results.iter().map(|(_, score)| *score).collect(),
                bm25_scores: vec![], // TODO: Return component scores from search_hybrid
                vector_scores: vec![],
                num_docs_considered: all_results.len() as u64,
                num_pages_accessed: 0,
            }))
        } else {
            Err(tonic::Status::not_found("Collection not found"))
        }
    }
    ```
