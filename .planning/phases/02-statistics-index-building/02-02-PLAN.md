    ```rust
    use unicode_segmentation::UnicodeWordIndices;
    use super::{Token, TokenStream, Tokenizer};
    
    pub struct UnicodeTokenizer;
    
    impl Default for UnicodeTokenizer {
        fn default() -> Self {
            Self::new()
        }
    }
    
    impl UnicodeTokenizer {
        pub fn new() -> Self {
            Self
        }
    }
    
    impl Tokenizer for UnicodeTokenizer {
        type TokenStream<'a> = UnicodeTokenStream<'a>;
    
        fn input<'a>(&self, text: &'a str) -> Self::TokenStream<'a> {
            UnicodeTokenStream {
                words: text.unicode_word_indices(),
                current_token: Token::new(String::new()),
            }
        }
    }
    
    pub struct UnicodeTokenStream<'a> {
        words: UnicodeWordIndices<'a>,
        current_token: Token,
    }
    
    impl<'a> TokenStream for UnicodeTokenStream<'a> {
        fn advance(&mut self) -> bool {
            if let Some((_, word)) = self.words.next() {
                self.current_token = Token::new(word.to_lowercase());
                true
            } else {
                false
            }
        }
    
        fn token(&self) -> Token {
            self.current_token.clone()
        }
    
        fn next(&mut self) -> Option<Token> {
            if self.advance() {
                Some(self.current_token.clone())
            } else {
                None
            }
        }
    }
    ```

    ```rust
    use std::collections::HashSet;
    use super::{Token, TokenStream};
    
    /// TokenStream wrapper that filters out stop words.
    pub struct StopWordFilter<T: TokenStream> {
        inner: T,
        stop_words: HashSet<String>,
        current_token: Token,
    }
    
    impl<T: TokenStream> StopWordFilter<T> {
        /// Create a new StopWordFilter wrapping an inner TokenStream.
        ///
        /// # Arguments
        /// * `inner` - The TokenStream to wrap
        /// * `stop_words` - Set of stop words to filter out
        pub fn new(inner: T, stop_words: HashSet<String>) -> Self {
            Self {
                inner,
                stop_words,
                current_token: Token::new(String::new()),
            }
        }
    
        /// Create with default English stop words.
        pub fn with_english_defaults(inner: T) -> Self {
            let mut stop_words = HashSet::new();
            // Common English stop words
            for word in &[
                "a", "an", "and", "are", "as", "at", "be", "by", "for", "from",
                "has", "he", "in", "is", "it", "its", "of", "on", "that", "the",
                "to", "was", "will", "with"
            ] {
                stop_words.insert(word.to_string());
            }
            Self::new(inner, stop_words)
        }
    
        fn advance_skipping_stop_words(&mut self) -> bool {
            while self.inner.advance() {
                let token = self.inner.token();
                if !self.stop_words.contains(&token.text) {
                    self.current_token = token;
                    return true;
                }
            }
            false
        }
    }
    
    impl<T: TokenStream> TokenStream for StopWordFilter<T> {
        fn advance(&mut self) -> bool {
            self.advance_skipping_stop_words()
        }
    
        fn token(&self) -> Token {
            self.current_token.clone()
        }
    
        fn next(&mut self) -> Option<Token> {
            if self.advance() {
                Some(self.current_token.clone())
            } else {
                None
            }
        }
    }
    ```

    ```rust
                stemmer: StemmingTokenizer::new(Algorithm::English),
                stop_words: Self::english_stop_words(),

                stop_words: HashSet::new(),
            }

        /// Tokenize text into final stemmed, filtered tokens.
        pub fn tokenize(&self, text: &str) -> Vec<String> {
            let mut tokens = Vec::new();
            let unicode_stream = self.unicode_tokenizer.input(text);
            let stemmed_stream = self.stemmer.input_from_stream(unicode_stream);
    ```

    Simpler approach for Pipeline:
    ```rust
    pub fn tokenize(&self, text: &str) -> Vec<String> {
        let mut tokens = Vec::new();
        
        // Step 1: Unicode word segmentation
        let mut unicode_stream = self.unicode_tokenizer.input(text);
        while let Some(token) = unicode_stream.next() {
            // Step 2: Stemming (re-tokenize each word through stemmer)
            let mut stem_stream = self.stemmer.input(&token.text);
            if let Some(stemmed) = stem_stream.next() {
                // Step 3: Stop word filtering
                if !self.stop_words.contains(&stemmed.text) {
                    tokens.push(stemmed.text);
                }
            }
        }
        tokens
    }
    ```
