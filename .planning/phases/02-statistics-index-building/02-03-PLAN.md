    ```rust
    use std::collections::HashMap;
    use anyhow::Result;
    use rkyv::{Archive, Serialize, Deserialize};
    
    /// Per-term statistics for serialization (DF only - TF is per-doc)
    #[derive(Archive, Serialize, Deserialize, Debug, Clone)]
    pub struct TermStat {
        pub term: String,
        pub doc_freq: u32,  // DF: number of docs containing this term
    }
    
    /// Per-user BM25 statistics for rkyv serialization
    #[derive(Archive, Serialize, Deserialize, Debug, Clone)]
    pub struct SerializableUserStats {
        pub num_docs: u32,
        pub total_terms: u64,
        pub avgdl: f64,
        pub doc_norms: Vec<u32>,  // Indexed by point_id (doc_norms[point_id] = dl)
        pub term_stats: Vec<TermStat>,  // DF for each term
    }
    
    impl SerializableUserStats {
        /// Convert from runtime UserBm25Stats to serializable form
        pub fn from_user_stats(user_stats: &UserBm25Stats, max_point_id: u32) -> Self {
            let num_docs = user_stats.doc_stats.len() as u32;
            let total_terms = user_stats.total_term_count;
            let avgdl = if num_docs > 0 {
                total_terms as f64 / num_docs as f64
            } else {
                0.0
            };
            
            // Build doc_norms array - indexed by point_id
            // Size to max_point_id + 1 for direct indexing
            let mut doc_norms = vec![0u32; (max_point_id + 1) as usize];
            for (point_id, doc_stats) in &user_stats.doc_stats {
                doc_norms[*point_id as usize] = doc_stats.term_count;
            }
            
            // Convert doc_frequencies HashMap to TermStat vec
            let term_stats: Vec<TermStat> = user_stats
                .doc_frequencies
                .iter()
                .map(|(term, df)| TermStat {
                    term: term.clone(),
                    doc_freq: *df,
                })
                .collect();
            
            Self {
                num_docs,
                total_terms,
                avgdl,
                doc_norms,
                term_stats,
            }
        }
    }
    
    /// Writer for persisting BM25 statistics per segment
    pub struct Bm25StatsWriter {
        base_directory: String,
    }
    
    impl Bm25StatsWriter {
        pub fn new(base_directory: String) -> Self {
            Self { base_directory }
        }
    
        /// Write statistics for all users to the bm25_stats/ subdirectory
        pub fn write(&self, collector: &MultiBm25StatsCollector, max_point_ids: &HashMap<u128, u32>) -> Result<()> {
            // Create bm25_stats subdirectory
            let stats_dir = format!("{}/bm25_stats", self.base_directory);
            std::fs::create_dir_all(&stats_dir)?;
            
            // Write stats for each user
            for user_id in collector.get_all_user_ids() {
                if let Some(user_stats) = collector.get_stats_for_user(user_id) {
                    let max_point_id = max_point_ids.get(&user_id).copied().unwrap_or(0);
                    let serializable = SerializableUserStats::from_user_stats(&user_stats, max_point_id);
                    
                    // Serialize with rkyv
                    let bytes = rkyv::to_bytes::<_, 256>(&serializable)
                        .map_err(|e| anyhow::anyhow!("rkyv serialization failed: {}", e))?;
                    
                    // Write to file: bm25_stats/{user_id}.rkyv
                    let file_path = format!("{}/{}.rkyv", stats_dir, user_id);
                    std::fs::write(&file_path, &bytes)?;
                }
            }
            
            Ok(())
        }
    }
    ```

    ```rust
    use std::collections::HashMap;
    use anyhow::{anyhow, Result};
    use rkyv::util::AlignedVec;
    
    use crate::bm25::stats_writer::{ArchivedSerializableUserStats, SerializableUserStats, TermStat};
    
    /// View of per-user BM25 statistics for query-time scoring
    pub struct UserBm25StatsView {
        pub num_docs: u32,
        pub total_terms: u64,
        pub avgdl: f64,
        pub doc_norms: Vec<u32>,  // Indexed by point_id
        pub term_df: HashMap<String, u32>,  // term -> DF
    }
    
    impl UserBm25StatsView {
        /// Get document length (dl) for a specific point_id
        pub fn get_doc_length(&self, point_id: u32) -> u32 {
            if (point_id as usize) < self.doc_norms.len() {
                self.doc_norms[point_id as usize]
            } else {
                0
            }
        }
    
        /// Get document frequency (DF) for a term
        pub fn get_term_df(&self, term: &str) -> u32 {
            self.term_df.get(term).copied().unwrap_or(0)
        }
    }
    
    /// Reader for loading BM25 statistics from disk
    pub struct Bm25StatsReader {
        base_directory: String,
        user_stats: HashMap<u128, UserBm25StatsView>,
    }
    
    impl Bm25StatsReader {
        /// Load all statistics from the bm25_stats/ subdirectory
        pub fn new(base_directory: String) -> Result<Self> {
            let stats_dir = format!("{}/bm25_stats", base_directory);
            let mut user_stats = HashMap::new();
            
            // Check if stats directory exists
            if std::path::Path::new(&stats_dir).exists() {
                // Read all .rkyv files
                for entry in std::fs::read_dir(&stats_dir)? {
                    let entry = entry?;
                    let path = entry.path();
                    
                    if let Some(filename) = path.file_stem().and_then(|s| s.to_str()) {
                        if let Ok(user_id) = filename.parse::<u128>() {
                            // Read and deserialize
                            let bytes = std::fs::read(&path)?;
                            let aligned = AlignedVec::<16>::from(bytes);
                            let archived = rkyv::access::<ArchivedSerializableUserStats, 16>(&aligned)
                                .map_err(|e| anyhow!("rkyv access failed: {:?}", e))?;
                            
                            // Convert to view
                            let view = UserBm25StatsView {
                                num_docs: archived.num_docs,
                                total_terms: archived.total_terms,
                                avgdl: archived.avgdl,
                                doc_norms: archived.doc_norms.to_vec(),
                                term_df: archived.term_stats
                                    .iter()
                                    .map(|ts| (ts.term.to_string(), ts.doc_freq))
                                    .collect(),
                            };
                            
                            user_stats.insert(user_id, view);
                        }
                    }
                }
            }
            
            Ok(Self {
                base_directory,
                user_stats,
            })
        }
    
        /// Get statistics for a specific user
        pub fn get_stats_for_user(&self, user_id: u128) -> Option<&UserBm25StatsView> {
            self.user_stats.get(&user_id)
        }
    
        /// Get list of all users with statistics
        pub fn get_user_ids(&self) -> Vec<u128> {
            self.user_stats.keys().copied().collect()
        }
    }
    ```
