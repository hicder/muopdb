       - Iterate inner_stats, collect all keys into Vec<u128>
    
    Add helper method to compute term frequencies:
    ```rust
    fn compute_term_frequencies(tokens: &[String]) -> HashMap<String, u32> {
        let mut freqs = HashMap::new();
        for token in tokens {
            *freqs.entry(token.clone()).or_insert(0) += 1;
        }
        freqs
    }
    ```
